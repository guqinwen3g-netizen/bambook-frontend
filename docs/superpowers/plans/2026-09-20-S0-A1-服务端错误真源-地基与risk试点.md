 * 路由错误真源守卫（S0-A Task 1）
 *
 * ── 为什么还要再加一条守卫 ─────────────────────────────
 * 既有棘轮 `routeError.test.ts` 的正则是
 *   /res\.status\([^)]*\)\.json\([^\n]*e\?\.message/g
 * 要求**同一行**出现 `e?.message`。但 risk/hr/dashboard/reporting 写的是
 *   const msg = e?.message;  ...  res.status(x).json({ error: { message: msg } })

const HERE = path.dirname(fileURLToPath(import.meta.url)); // server/src/shared/__tests__
const SRC = path.resolve(HERE, '..', '..'); // server/src

/** 这两个文件本身就是真源/组合根，不参与判据 */
const SKIP_FILES = new Set(['index.ts', 'lib/asyncHandler.ts']);

function stripComments(src: string): string {
  return src
    .replace(/\/\*[\s\S]*?\*\//g, '')
    .replace(/(^|[^:])\/\/[^\n]*/g, '$1'); // 前一个字符不是 ':'，避免误伤 https://
}

/** 纯函数判据：给定相对路径与**已剥注释**的源码，是否违规。 */
export function isNoTruthRoute(rel: string, strippedCode: string): boolean {
  if (SKIP_FILES.has(rel)) return false;
  if (!/Router\(\)/.test(strippedCode)) return false; // 只看定义路由的文件
  const respondsInCatch = /catch[\s\S]{0,600}?res\.status\([^)]*\)\.json\(/.test(strippedCode);
  if (!respondsInCatch) return false;
  const importsTruth =
    /from '[.\w/]*shared\/routeError'/.test(strippedCode) ||
    /from '[.\w/]*lib\/asyncHandler'/.test(strippedCode);
  return !importsTruth;
}

function scanNoTruthRoutes(): string[] {
  const out: string[] = [];
  const walk = (dir: string) => {
    for (const ent of fs.readdirSync(dir, { withFileTypes: true })) {
      if (ent.name === '__tests__' || ent.name === 'node_modules') continue;
      const abs = path.join(dir, ent.name);
      if (ent.isDirectory()) {
        walk(abs);
      } else if (ent.name.endsWith('.ts') && !ent.name.endsWith('.test.ts')) {
        const rel = path.relative(SRC, abs).split(path.sep).join('/');
        if (isNoTruthRoute(rel, stripComments(fs.readFileSync(abs, 'utf8')))) out.push(rel);
      }
    }
  };
  walk(SRC);
  return out.sort();
}

/**
 * resolveError 语义钉子（S0-A Task 2）
 * 三条不可让的语义：
 *   1. 业务错误（自带 code+statusCode）→ 原文透传，用自己的状态码
 *   2. 未预期异常 → 注册表文案；注册表没这个词 → 通用文案。**原始 message 一个字都不许出现。**
 *   3. details / gateDetails 必须随附（出运门禁 409 的前端契约）
 */
import { describe, expect, it } from 'vitest';
import {
  ERROR_CODES, GENERIC_ROUTE_ERROR_MESSAGE, lookupErrorCode, resolveError,
} from '../errorCodes';

describe('resolveError', () => {
  it('业务错误：透传 code/message，且用它自己的 statusCode', () => {
    const e = Object.assign(new Error('币种不一致'), { code: 'CURRENCY_MISMATCH', statusCode: 400 });
    expect(resolveError(e, 'CREATE_FAILED')).toEqual({
      status: 400, code: 'CURRENCY_MISMATCH', message: '币种不一致',
    });
  });

  it('业务错误带 gateDetails → details 必须随附', () => {
    const details = [{ orderId: 'o1', missing: ['船样确认'] }];
    const e = Object.assign(new Error('出运门禁未通过'), {
      code: 'GATE_BLOCKED', statusCode: 409, gateDetails: details,
    });
    const r = resolveError(e, 'SHIP_FAILED');
    expect(r.status).toBe(409);
    expect(r.code).toBe('GATE_BLOCKED');
    expect(r.details).toEqual(details);
  });

  it('未预期异常：Prisma 原文一个字都不许进响应', () => {
    const raw = 'PrismaClientKnownRequestError: Invalid `prisma.riskAlert.findMany()` invocation in /Users/panda1/bambook-main-api/server/src/risk/riskService.ts:244: Unknown argument `deletedAts`';
    const r = resolveError(new Error(raw), 'ALERT_LIST_FAILED');
    expect(JSON.stringify(r)).not.toContain('Prisma');
    expect(JSON.stringify(r)).not.toContain('bambook-main-api');
    expect(JSON.stringify(r)).not.toContain('deletedAts');
    expect(r.status).toBe(500);
    expect(r.code).toBe('ALERT_LIST_FAILED');
  });

  it('注册表里有的 fallback code → 用注册表文案（不是通用文案）', () => {
    const r = resolveError(new Error('内部细节'), 'OVERVIEW_FAILED');
    expect(r.message).toBe(ERROR_CODES.OVERVIEW_FAILED.message);
    expect(r.message).not.toContain('内部细节');
  });

  it('注册表里没有的 fallback code → 通用文案 + 500', () => {
    const r = resolveError(new Error('内部细节'), 'NOT_A_REAL_CODE');
    expect(r.status).toBe(500);
    expect(r.message).toBe(GENERIC_ROUTE_ERROR_MESSAGE);
  });

 *
 * 为什么要有这张表：全仓曾把「给用户看什么话」当成每个 catch 点现写的手艺活，

export interface ResolvedRouteError {
  status: number;
  code: string;
  message: string;
  details?: unknown;
}

/**
 * 把任意被捕获的值解析成「可安全外发」的错误。
 * 日志由调用方负责（本函数不记日志，也不吞原始错误 —— 保持既有日志行为不变）。
 */
export function resolveError(e: unknown, fallbackCode: string): ResolvedRouteError {
  const err = e as {
    code?: unknown; statusCode?: unknown; message?: unknown;
    details?: unknown; gateDetails?: unknown;
  } | null | undefined;

  // 1) 业务错误：本仓约定 code(string) + statusCode(number) 同时存在
  if (typeof err?.code === 'string' && typeof err.statusCode === 'number') {
    const details = err.details ?? err.gateDetails;
    return {
      status: err.statusCode,
      code: err.code,
      message: String(err.message ?? err.code),
      ...(details === undefined ? {} : { details }),
    };
  }

  // 2) 未预期：按 fallback 码查表出文案；查不到 → 通用文案
  const spec = lookupErrorCode(fallbackCode);
  return {
    status: spec?.status ?? 500,
    code: fallbackCode,
    message: spec?.message ?? GENERIC_ROUTE_ERROR_MESSAGE,
  };
}
